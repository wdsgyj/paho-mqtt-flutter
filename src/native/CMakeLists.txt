cmake_minimum_required(VERSION 3.22)
project(mqtt_sdk VERSION 0.1.0 LANGUAGES C)

set(PAHO_SOURCE_DIR "" CACHE PATH "Paho source tree with the connected-session extension")
if(NOT EXISTS "${PAHO_SOURCE_DIR}/src/MQTTAsync.c")
  message(FATAL_ERROR "Set PAHO_SOURCE_DIR to the Paho source tree")
endif()
option(MQTT_WITH_TLS "Link the TLS-enabled Paho library" ON)
option(MQTT_BUILD_HOST_CHECK "Build a host-only ABI verification program" OFF)

if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" AND NOT MQTT_BUILD_HOST_CHECK)
  message(FATAL_ERROR "Only Android and iOS are product targets. Use MQTT_BUILD_HOST_CHECK for local ABI checks.")
endif()
if(ANDROID)
  if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Only ANDROID_ABI=arm64-v8a is supported")
  endif()
  # The NDK legacy toolchain uses CMAKE_SYSTEM_VERSION=1, not the API level.
  string(REGEX REPLACE "^android-" "" MQTT_ANDROID_API "${ANDROID_PLATFORM}")
  if(NOT MQTT_ANDROID_API MATCHES "^[0-9]+$" OR MQTT_ANDROID_API LESS 21)
    message(FATAL_ERROR "Android API must be at least 21")
  endif()
endif()

# Replace CMake's default Release -O3, including for the Paho dependency.
set(CMAKE_C_FLAGS_RELEASE "-Os -g -DNDEBUG" CACHE STRING "Release C flags" FORCE)
set(CMAKE_CXX_FLAGS_RELEASE "-Os -g -DNDEBUG" CACHE STRING "Release C++ flags" FORCE)
# NDK also creates normal variables that shadow cache values (CMP0126).
set(CMAKE_C_FLAGS_RELEASE "-Os -g -DNDEBUG")
set(CMAKE_CXX_FLAGS_RELEASE "-Os -g -DNDEBUG")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)

set(PAHO_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(PAHO_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(PAHO_WITH_SSL ${MQTT_WITH_TLS} CACHE BOOL "" FORCE)
set(PAHO_WITH_LIBRESSL OFF CACHE BOOL "" FORCE)
set(PAHO_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(PAHO_BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
set(PAHO_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(PAHO_ENABLE_CPACK OFF CACHE BOOL "" FORCE)
if(MQTT_WITH_TLS)
  set(OPENSSL_USE_STATIC_LIBS TRUE)
  if((ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS") AND
     (NOT OPENSSL_INCLUDE_DIR OR NOT OPENSSL_SSL_LIBRARY OR NOT OPENSSL_CRYPTO_LIBRARY))
    message(FATAL_ERROR "Supply target OpenSSL include, libssl.a and libcrypto.a paths")
  endif()
endif()
# Build the MQTT 5 fixes in an isolated overlay, leaving the dependency checkout untouched.
find_program(MQTT_GO_EXECUTABLE go REQUIRED)
file(GLOB MQTT_GO_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/*.go")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${MQTT_GO_SOURCES})
set(MQTT_PAHO_ORIGINAL "${PAHO_SOURCE_DIR}")
set(MQTT_PAHO_OVERLAY "${CMAKE_CURRENT_BINARY_DIR}/paho-sdk-source")
execute_process(COMMAND "${MQTT_GO_EXECUTABLE}" run . overlay
  "${MQTT_PAHO_ORIGINAL}" "${MQTT_PAHO_OVERLAY}"
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts" COMMAND_ERROR_IS_FATAL ANY)
set(PAHO_SOURCE_DIR "${MQTT_PAHO_OVERLAY}")
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/core" "${PAHO_SOURCE_DIR}/src")
add_subdirectory("${PAHO_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/paho" EXCLUDE_FROM_ALL)

if(ANDROID OR MQTT_BUILD_HOST_CHECK)
  add_library(mqtt_sdk SHARED core/mqtt_api.c)
else()
  add_library(mqtt_sdk STATIC core/mqtt_api.c)
endif()
# Dart's official DL shim resolves VM functions from NativeApi.initializeApiDLData.
find_program(MQTT_DART_EXECUTABLE dart)
if(MQTT_DART_EXECUTABLE)
  get_filename_component(MQTT_DART_BIN "${MQTT_DART_EXECUTABLE}" DIRECTORY)
endif()
find_path(DART_SDK_INCLUDE_DIR dart_api_dl.h
  HINTS "$ENV{DART_SDK}/include" "${MQTT_DART_BIN}/../include"
        "${MQTT_DART_BIN}/cache/dart-sdk/include"
  NO_CMAKE_FIND_ROOT_PATH)
if(NOT DART_SDK_INCLUDE_DIR)
  message(FATAL_ERROR "Set DART_SDK_INCLUDE_DIR to the Dart SDK include directory")
endif()
set_source_files_properties("${DART_SDK_INCLUDE_DIR}/dart_api_dl.c" PROPERTIES COMPILE_OPTIONS "-Wno-unused-parameter")
target_sources(mqtt_sdk PRIVATE core/mqtt_dart_bridge.c "${DART_SDK_INCLUDE_DIR}/dart_api_dl.c")
target_include_directories(mqtt_sdk PRIVATE "${DART_SDK_INCLUDE_DIR}")
target_include_directories(mqtt_sdk PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_definitions(mqtt_sdk PRIVATE MQTT_SDK_DIAGNOSTICS=1)
target_sources(mqtt_sdk PRIVATE errors/mqtt_diagnostics.c)
target_include_directories(mqtt_sdk PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/errors")
if(MQTT_WITH_TLS)
  set(MQTT_DIAG_TARGETS paho-mqtt3as-static common_ssl_obj_static)
else()
  set(MQTT_DIAG_TARGETS paho-mqtt3a-static common_obj_static)
endif()
file(READ "${PAHO_SOURCE_DIR}/src/MQTTAsyncUtils.c" MQTT_PAHO_DIAG_SOURCE)
if(NOT MQTT_PAHO_DIAG_SOURCE MATCHES "mqtt_sdk_connect_failed")
  message(FATAL_ERROR "Apply patches/paho-diagnostics.patch before building")
endif()
foreach(t IN LISTS MQTT_DIAG_TARGETS)
  target_compile_definitions(${t} PRIVATE MQTT_SDK_DIAGNOSTICS=1)
  target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/errors" "${CMAKE_CURRENT_SOURCE_DIR}/include")
endforeach()
target_compile_features(mqtt_sdk PRIVATE c_std_11)
target_compile_options(mqtt_sdk PRIVATE -Wall -Wextra -Werror)
if(MQTT_WITH_TLS)
  # Only this SDK opts into the small Paho trust hook; standalone Paho is unchanged.
  file(READ "${PAHO_SOURCE_DIR}/src/SSLSocket.c" MQTT_PAHO_SSL_SOURCE)
  if(NOT MQTT_PAHO_SSL_SOURCE MATCHES "MQTT_SDK_PLATFORM_TRUST")
    message(FATAL_ERROR "Apply patches/paho-platform-trust.patch before building the TLS SDK")
  endif()
  target_compile_definitions(paho-mqtt3as-static PRIVATE MQTT_SDK_PLATFORM_TRUST=1)
  target_include_directories(paho-mqtt3as-static PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tls")
  target_sources(mqtt_sdk PRIVATE tls/mqtt_tls.c)
  target_include_directories(mqtt_sdk PRIVATE "${OPENSSL_INCLUDE_DIR}")
  if(APPLE)
    target_link_libraries(mqtt_sdk PRIVATE "-framework Security" "-framework CoreFoundation")
  endif()
  target_compile_definitions(mqtt_sdk PRIVATE MQTT_SDK_TLS=1)
  target_link_libraries(mqtt_sdk PRIVATE paho-mqtt3as-static)
else()
  target_link_libraries(mqtt_sdk PRIVATE paho-mqtt3a-static)
endif()
if(ANDROID)
  target_sources(mqtt_sdk PRIVATE jni/mqtt_bootstrap.c)
  target_link_options(mqtt_sdk PRIVATE
    "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/jni/exports.map"
    "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")
  set_property(TARGET mqtt_sdk APPEND PROPERTY LINK_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/jni/exports.map")
endif()
if(APPLE AND MQTT_BUILD_HOST_CHECK AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  # The function table is the sole host FFI entry point. Keep dependency symbols
  # private to this dylib, including default-visible symbols from static OpenSSL.
  target_link_options(mqtt_sdk PRIVATE "-Wl,-exported_symbol,_mqtt_get_api")
endif()
if(MQTT_BUILD_HOST_CHECK)
  enable_testing()
  add_executable(mqtt_abi_check tests/abi_check.c)
  target_link_libraries(mqtt_abi_check PRIVATE mqtt_sdk)
  add_test(NAME mqtt_abi_check COMMAND mqtt_abi_check)
  add_executable(mqtt_notification_check tests/notification_check.c)
  target_link_libraries(mqtt_notification_check PRIVATE mqtt_sdk)
  add_test(NAME mqtt_notification_check COMMAND mqtt_notification_check)
  set_tests_properties(mqtt_notification_check PROPERTIES TIMEOUT 20)
  add_executable(mqtt_error_retry_check tests/error_retry_check.c)
  target_link_libraries(mqtt_error_retry_check PRIVATE mqtt_sdk)
  add_test(NAME mqtt_error_retry_check COMMAND mqtt_error_retry_check)
  set_tests_properties(mqtt_error_retry_check PROPERTIES TIMEOUT 40)
  add_executable(mqtt_v2_check tests/v2_check.c)
  target_link_libraries(mqtt_v2_check PRIVATE mqtt_sdk)
  add_executable(mqtt_error_matrix_check tests/error_matrix_check.c)
  target_link_libraries(mqtt_error_matrix_check PRIVATE mqtt_sdk)
  add_executable(mqtt_runtime_check tests/runtime_check.c)
  target_link_libraries(mqtt_runtime_check PRIVATE mqtt_sdk)
endif()
